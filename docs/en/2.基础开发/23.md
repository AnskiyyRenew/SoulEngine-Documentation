# Beat

This library is suitable for scenarios that require precise beat synchronization, such as rhythm games, audio visualization, or interactive media.

---

## Main modules and methods

### Configuration
- `SetBPM(new_bpm)`: set the BPM value (must be positive).
- `SetCycleLength(length)`: set the beat cycle length (for example, 4 beats per measure).

### State retrieval
- `GetBeat()`: returns the current beat integer part and fractional part.
- `GetBar()`: returns the current bar number and the beat position within the cycle.

### Event handling
- `RegisterEvent(event_type, handler)`: register an event callback (supports `beat` and `reset`).

### Beat control
- `SetBeat(beat)`: directly set the current beat (triggers reset events).
- `Reset()`: reset the beat to 0 while keeping the BPM setting.

### State update
- `Update(dt)`: update the beat state based on time delta in seconds, automatically triggering beat events and cycle detection.

### Condition checks
- `OnBeat(target_beat)`: checks whether the target beat has been reached (supports relative beat detection within a cycle).

```lua
-- Import the Beat manager
local Beat = require("Beat.lua")

-- Example 1: basic beat tracking and event triggers
function setupBasicBeatTracking()
    -- Set BPM to 120 (common pop tempo)
    Beat.SetBPM(120)
    
    -- Register a beat event: triggers on every whole beat
    Beat.RegisterEvent("beat", function(current_beat)
        print(string.format("Beat %d triggered!", current_beat))
    end)
    
    -- Register a reset event
    Beat.RegisterEvent("reset", function(beat)
        print(string.format("Beat reset to: %.2f", beat))
    end)
end

-- Example 2: rhythm-synced game logic
function gameBeatExample()
    Beat.SetBPM(140)  -- set a faster tempo
    Beat.SetCycleLength(4)  -- 4-beat measure
    
    -- Simulated game update loop (dt is frame delta)
    function gameUpdate(dt)
        Beat.Update(dt)  -- update beat state
        
        -- Detect a specific beat point (like the first beat of the measure)
        if Beat.OnBeat(0) then
            spawnEnemy()  -- spawn an enemy on the strong beat
        end
        
        -- Detect the third beat inside the cycle
        if Beat.OnBeat(2) then
            triggerSpecialEffect()  -- trigger an effect
        end
        
        -- Get beat info for UI display
        local integer_beat, fractional_beat = Beat.GetBeat()
        updateBeatUI(integer_beat, fractional_beat)
    end
end

-- Example 3: audio visualization
function audioVisualizationExample()
    Beat.SetBPM(90)
    Beat.SetCycleLength(8)  -- 8 beats per phrase
    
    function onAudioFrame(dt)
        Beat.Update(dt)
        
        local current_bar, bar_beat = Beat.GetBar()
        
        -- Adjust visual intensity based on bar position
        if bar_beat < 2 then
            setVisualizerIntensity(0.5)  -- weak beat phase
        else
            setVisualizerIntensity(1.0)  -- strong beat phase
        end
        
        -- Highlight the first beat of each bar
        if Beat.OnBeat(0) then
            triggerBarHighlight()
        end
    end
end

-- Example 4: manual beat control (debug or special scene)
function manualControlDemo()
    -- Jump directly to a specific beat
    Beat.SetBeat(16.5)  -- jump to beat 16.5
    
    -- Reset the entire beat system
    Beat.Reset()
end

-- Run the example
setupBasicBeatTracking()

-- Simulate time progression (called every frame)
for i = 1, 100 do
    Beat.Update(1/60)  -- assuming 60 FPS
end
```
